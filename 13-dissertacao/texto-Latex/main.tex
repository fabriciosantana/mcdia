%% Compatibilidade com biblatex-abnt anterior ao 4.0 (ex.: TeX Live 2023), que
%% nao declara a opcao `accite' usada pela classe do IDP. A partir do 4.0 a opcao
%% e declarada pelo proprio estilo, e uma segunda declaracao aborta a compilacao
%% com "Package biblatex Error: Option 'accite' already defined".
%% A decisao e tomada pelo conteudo do abnt.cbx instalado, de modo que o mesmo
%% main.tex compile tanto no TeX Live 2023 (Ubuntu 24.04) quanto no TeX Live 2026.
\makeatletter
\newcommand*{\idpShimAccite}{%          % nome sem `@' nem `_': legivel dentro do expl3
  \AddToHook{package/biblatex/before}{%
    \define@boolkey{blx@opt@pre}{accite}[true]{}%
  }%
}
\makeatother
\ExplSyntaxOn
\file_if_exist:nTF { abnt.cbx }
  {
    \file_get:nnN { abnt.cbx } { } \l_tmpa_tl
    \str_if_in:NnF \l_tmpa_tl { accite } { \idpShimAccite }
  }
  { \idpShimAccite }
\ExplSyntaxOff
